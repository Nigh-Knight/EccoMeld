   ```javascript
   import { readdirSync, readFileSync, writeFileSync } from "fs";
   import { join } from "path";

   const outDir = new URL("../out", import.meta.url).pathname;

   function stripCrossorigin(html) {
     return html
       .replace(/\s+crossorigin="[^"]*"/g, "")
       .replace(/\s+crossorigin(?=[>\s/])/g, "");
   }

   function processDir(dir) {
     for (const entry of readdirSync(dir, { withFileTypes: true })) {
       const fullPath = join(dir, entry.name);
       if (entry.isDirectory()) {
         processDir(fullPath);
       } else if (entry.name.endsWith(".html")) {
         const original = readFileSync(fullPath, "utf8");
         const stripped = stripCrossorigin(original);
         if (stripped !== original) writeFileSync(fullPath, stripped);
       }
     }
   }

   processDir(outDir);
   console.log("strip-crossorigin: done");
   ```
   This recursively processes all HTML files in eccopath/out/ and removes `crossorigin` attributes. Belt-and-suspenders defense against Next.js #61210.

Add two Gradle tasks to `app/build.gradle.kts`, placed BEFORE the `android {}` block (after the `plugins {}` and `localProperties` blocks, before `android {`). Per D-04, these automate the full EccoPath build pipeline.

1. Register a `buildEccoPath` Exec task:
   ```kotlin
   val buildEccoPath by tasks.registering(Exec::class) {
       description = "Build EccoPath Next.js static export for Android"
       group = "eccopath"
       workingDir = file("${rootProject.projectDir}/eccopath")
       environment("NEXT_CONFIG_FILE", "next.config.android.ts")
       commandLine("node", "${rootProject.projectDir}/eccopath/node_modules/.bin/next", "build")
       doLast {
           exec {
               workingDir = file("${rootProject.projectDir}/eccopath")
               commandLine("node", "scripts/strip-crossorigin.mjs")
           }
       }
       inputs.dir("${rootProject.projectDir}/eccopath/lib")
       inputs.dir("${rootProject.projectDir}/eccopath/app")
       inputs.file("${rootProject.projectDir}/eccopath/next.config.android.ts")
       outputs.dir("${rootProject.projectDir}/eccopath/out")
   }
   ```
   Key details:
   - Uses `NEXT_CONFIG_FILE` env var (not `--config` flag) to override config -- this is the reliable approach per research
   - Uses `node eccopath/node_modules/.bin/next` directly to avoid PATH dependency
   - `doLast` runs the crossorigin strip script after the Next.js build completes
   - `inputs`/`outputs` enable Gradle incremental build caching

2. Register a `copyEccoPathAssets` Copy task:
   ```kotlin
   val copyEccoPathAssets by tasks.registering(Copy::class) {
       dependsOn(buildEccoPath)
       from("${rootProject.projectDir}/eccopath/out")
       into("${projectDir}/src/main/assets/eccopath")
   }
   ```

3. Wire `copyEccoPathAssets` into the Android build by adding after the `android {}` block closes:
   ```kotlin
   tasks.named("preBuild").configure {
       dependsOn(copyEccoPathAssets)
   }
   ```
   This ensures every Android build automatically builds and bundles EccoPath assets.
